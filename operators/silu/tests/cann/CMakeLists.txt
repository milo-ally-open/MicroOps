cmake_minimum_required(VERSION 3.21)
project(silu_cann_tests LANGUAGES CXX)
include(CTest)

get_filename_component(SILU_OPERATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ASCEND_TOOLKIT_ROOT "$ENV{ASCEND_HOME_PATH}")
if(NOT ASCEND_TOOLKIT_ROOT)
    set(ASCEND_TOOLKIT_ROOT "$ENV{ASCEND_TOOLKIT_HOME}")
endif()
if(NOT ASCEND_TOOLKIT_ROOT)
    set(ASCEND_TOOLKIT_ROOT "/usr/local/Ascend/ascend-toolkit/latest")
endif()
set(ASCEND_INCLUDE_DIR "${ASCEND_TOOLKIT_ROOT}/aarch64-linux/include")
set(ASCEND_LIB_DIR "${ASCEND_TOOLKIT_ROOT}/aarch64-linux/lib64")
add_subdirectory("${SILU_OPERATOR_ROOT}/src/cann" "${CMAKE_CURRENT_BINARY_DIR}/operator_cann")
set(SILU_CANN_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/operator_cann/libsilu_cann.so")

set(OPS_TEST_CORRECTNESS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/correctness/silu_cann_test.cpp" CACHE FILEPATH "CANN correctness source")
set(OPS_TEST_MEMORY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/memory/silu_cann_memory_eval.cpp" CACHE FILEPATH "CANN memory evaluation source")
set(OPS_TEST_PERFORMANCE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/performance/silu_cann_benchmark.cpp" CACHE FILEPATH "CANN benchmark source")
set(SILU_CANN_MEMORY_RESULTS "${SILU_OPERATOR_ROOT}/tests/cann/memory/results/memory.csv" CACHE FILEPATH "CANN memory record CSV")
get_filename_component(_memory_dir "${SILU_CANN_MEMORY_RESULTS}" DIRECTORY)
file(MAKE_DIRECTORY "${_memory_dir}")

function(silu_cann_test target source)
    add_executable(${target} "${source}")
    target_include_directories(${target} PRIVATE "${SILU_OPERATOR_ROOT}/src/cann/include" "${ASCEND_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE "${SILU_CANN_LIBRARY}" "${ASCEND_LIB_DIR}/libascendcl.so")
    target_link_options(${target} PRIVATE "-Wl,-rpath,${ASCEND_LIB_DIR}" "-Wl,-rpath,${CMAKE_CURRENT_BINARY_DIR}/operator_cann")
    target_compile_definitions(${target} PRIVATE
        SILU_CANN_KERNEL_FILE="${OPS_KERNEL_FILE}" SILU_CANN_HOST_FILE="${OPS_HOST_FILE}"
        SILU_CANN_RUNTIME_VERSION="${SILU_CANN_RUNTIME_VERSION}")
    add_dependencies(${target} silu_cann)
endfunction()

silu_cann_test(silu_cann_correctness "${OPS_TEST_CORRECTNESS_SOURCE}")
add_test(NAME silu_cann_correctness COMMAND silu_cann_correctness)
set_tests_properties(silu_cann_correctness PROPERTIES SKIP_RETURN_CODE 77)

add_executable(silu_cann_memory_eval "${OPS_TEST_MEMORY_SOURCE}")
target_include_directories(silu_cann_memory_eval PRIVATE "${ASCEND_INCLUDE_DIR}")
target_link_libraries(silu_cann_memory_eval PRIVATE "${ASCEND_LIB_DIR}/libascendcl.so")
target_link_options(silu_cann_memory_eval PRIVATE "-Wl,-rpath,${ASCEND_LIB_DIR}")
add_test(NAME silu_cann_memory_eval COMMAND silu_cann_memory_eval "${SILU_CANN_MEMORY_RESULTS}" 65536)
set_tests_properties(silu_cann_memory_eval PROPERTIES SKIP_RETURN_CODE 77)

option(OPS_BUILD_PERFORMANCE "Build CANN performance evaluator" OFF)
if(OPS_BUILD_PERFORMANCE)
    silu_cann_test(silu_cann_benchmark "${OPS_TEST_PERFORMANCE_SOURCE}")
endif()
