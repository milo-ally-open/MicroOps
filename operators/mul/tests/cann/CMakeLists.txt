cmake_minimum_required(VERSION 3.21)
project(mul_cann_tests LANGUAGES CXX)
include(CTest)

get_filename_component(MUL_OPERATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(ASCEND_TOOLKIT_ROOT "$ENV{ASCEND_HOME_PATH}")
if(NOT ASCEND_TOOLKIT_ROOT)
    set(ASCEND_TOOLKIT_ROOT "$ENV{ASCEND_TOOLKIT_HOME}")
endif()
if(NOT ASCEND_TOOLKIT_ROOT)
    set(ASCEND_TOOLKIT_ROOT "/usr/local/Ascend/ascend-toolkit/latest")
endif()
set(ASCEND_INCLUDE_DIR "${ASCEND_TOOLKIT_ROOT}/aarch64-linux/include")
set(ASCEND_LIB_DIR "${ASCEND_TOOLKIT_ROOT}/aarch64-linux/lib64")
add_subdirectory("${MUL_OPERATOR_ROOT}/src/cann" "${CMAKE_CURRENT_BINARY_DIR}/operator_cann")
set(MUL_CANN_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/operator_cann/libmul_cann.so")

set(OPS_TEST_CORRECTNESS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/test/mul_cann_test.cpp" CACHE FILEPATH "CANN correctness source")
set(OPS_TEST_MEMORY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/test/mul_cann_memory_eval.cpp" CACHE FILEPATH "CANN memory evaluation source")
set(OPS_TEST_BENCH_SOURCE "${CMAKE_CURRENT_LIST_DIR}/bench/mul_cann_bench.cpp" CACHE FILEPATH "CANN benchmark source")
set(MUL_CANN_MEMORY_RESULTS "${MUL_OPERATOR_ROOT}/tests/cann/test/results/memory.csv" CACHE FILEPATH "CANN memory record CSV")
get_filename_component(_memory_dir "${MUL_CANN_MEMORY_RESULTS}" DIRECTORY)
file(MAKE_DIRECTORY "${_memory_dir}")

function(mul_cann_test target source)
    add_executable(${target} "${source}")
    target_include_directories(${target} PRIVATE "${MUL_OPERATOR_ROOT}/src/cann/include" "${ASCEND_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE "${MUL_CANN_LIBRARY}" "${ASCEND_LIB_DIR}/libascendcl.so")
    target_link_options(${target} PRIVATE "-Wl,-rpath,${ASCEND_LIB_DIR}" "-Wl,-rpath,${CMAKE_CURRENT_BINARY_DIR}/operator_cann")
    target_compile_definitions(${target} PRIVATE
        MUL_CANN_KERNEL_FILE="${OPS_KERNEL_FILE}" MUL_CANN_HOST_FILE="${OPS_HOST_FILE}"
        MUL_CANN_RUNTIME_VERSION="${MUL_CANN_RUNTIME_VERSION}")
    add_dependencies(${target} mul_cann)
endfunction()

mul_cann_test(mul_cann_correctness "${OPS_TEST_CORRECTNESS_SOURCE}")
add_test(NAME mul_cann_correctness COMMAND mul_cann_correctness)
set_tests_properties(mul_cann_correctness PROPERTIES SKIP_RETURN_CODE 77)

add_executable(mul_cann_memory_eval "${OPS_TEST_MEMORY_SOURCE}")
target_include_directories(mul_cann_memory_eval PRIVATE "${ASCEND_INCLUDE_DIR}")
target_link_libraries(mul_cann_memory_eval PRIVATE "${ASCEND_LIB_DIR}/libascendcl.so")
target_link_options(mul_cann_memory_eval PRIVATE "-Wl,-rpath,${ASCEND_LIB_DIR}")
add_test(NAME mul_cann_memory_eval COMMAND mul_cann_memory_eval "${MUL_CANN_MEMORY_RESULTS}" 65536)
set_tests_properties(mul_cann_memory_eval PROPERTIES SKIP_RETURN_CODE 77)

option(OPS_BUILD_BENCH "Build CANN bench evaluator" OFF)
if(OPS_BUILD_BENCH)
    mul_cann_test(mul_cann_bench "${OPS_TEST_BENCH_SOURCE}")
endif()
